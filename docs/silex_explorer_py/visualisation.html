13def replicate_scientific_objects(df, csv_filepath=None):
14    """
15    Group scientific objects based on variable combinations and optionally
16    generate a CSV summary of the resulting groups.
17
18    Parameters
19    ----------
20    df : pandas.DataFrame
21        Input DataFrame containing scientific objects metadata.
22        Must include at least the columns 'URI' and 'Name'.
23
24    csv_filepath : str, optional
25        Path to the CSV file where the group summary will be saved.
26        If None, no CSV file is generated.
27        Parent directories are automatically created if they do not exist.
28
29    Returns
30    -------
31    dict
32        Dictionary where keys are group identifiers and values are DataFrames
33        corresponding to each group.
34    """
35
36    # Exclude 'URI' and 'Name' columns from the comparison
37    columns_to_compare = [
38        col for col in df.columns if col not in ['URI', 'Name']
39    ]
40    
41    # Exclude columns where all values are identical
42    columns_to_compare = [
43        col for col in columns_to_compare
44        if df[col].nunique(dropna=False) > 1
45    ]
46
47    # Convert all values in relevant columns to string to avoid type issues
48    df[columns_to_compare] = df[columns_to_compare].astype(str)
49    # Function to remove trailing NaNs from the group identifier
50    def remove_trailing_nans_from_identifier(group_identifier):
51        """
52        Remove trailing 'nan' values only at the end of the group identifier.
53        """
54        parts = group_identifier.split('_')
55        while parts and parts[-1] == 'nan':
56            parts.pop()
57        return '_'.join(parts)
58
59    # Create a unique identifier for each row based on selected columns
60    df['group_identifier'] = df[columns_to_compare].agg('_'.join, axis=1)
61
62    # Clean group identifiers by removing trailing NaNs
63    df['group_identifier'] = df['group_identifier'].apply(
64        remove_trailing_nans_from_identifier
65    )
66
67    # Assign a default identifier if the result is empty
68    df['group_identifier'] = df['group_identifier'].apply(
69        lambda x: x if x != '' else 'NaN_group'
70    )
71
72    # Dictionary to store grouped DataFrames
73    group_dict = {}
74
75    # List to store group summary information
76    group_summary = []
77
78    # Group DataFrame by group identifier
79    for group, group_df in df.groupby('group_identifier'):
80        # Store group DataFrame without the group identifier column
81        group_dict[group] = group_df.drop(columns=['group_identifier'])
82        # Store group name and number of elements
83        group_summary.append([group, len(group_df)])
84
85    # Save group summary to CSV if requested
86    if csv_filepath:
87        # Ensure output directory exists
88        if os.path.dirname(csv_filepath):
89            os.makedirs(os.path.dirname(csv_filepath), exist_ok=True)
90
91        # Create and save the summary DataFrame
92        group_summary_df = pd.DataFrame(
93            group_summary,
94            columns=['Group', 'Number of Elements']
95        )
96        group_summary_df.to_csv(csv_filepath, index=False)
97        print(f"✅ Group summary has been saved to '{csv_filepath}'.")
98
99    return group_dict
